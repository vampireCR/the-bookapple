import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; import QRCode from "npm:qrcode@1.5.4"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", };
